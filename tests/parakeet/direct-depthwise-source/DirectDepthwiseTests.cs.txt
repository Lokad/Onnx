using System.Diagnostics;
using System.Reflection;
using System.Runtime.InteropServices;
using System.Runtime.Intrinsics.X86;
using System.Runtime.Loader;
using System.Security.Cryptography;
using System.Text.Json;

namespace Lokad.Onnx.Backend.Tests;

public class DirectDepthwiseTests
{
    static readonly string OriginalDirectory = Environment.GetEnvironmentVariable("DEPTHWISE_ORIGINAL")
        ?? throw new InvalidOperationException("Original M78 runtime is required.");
    static readonly Assembly Original = LoadOriginal();
    static readonly Type OriginalTensor = Original.GetType("Lokad.Onnx.Tensor`1")!.MakeGenericType(typeof(float));
    static readonly Type OriginalDense = Original.GetType("Lokad.Onnx.DenseTensor`1")!.MakeGenericType(typeof(float));
    static readonly Type OriginalOptions = Original.GetType("Lokad.Onnx.TensorExecutionOptions")!;
    static readonly MethodInfo OriginalConv = OriginalTensor.GetMethods().Single(m => m.Name == "Conv2D"
        && m.GetParameters().Length == 9 && m.GetParameters()[0].ParameterType == OriginalTensor
        && m.GetParameters()[3].ParameterType == typeof(int[]));

    static Assembly LoadOriginal()
    {
        var context = new AssemblyLoadContext("depthwise-original", isCollectible: false);
        context.Resolving += (owner, name) =>
        {
            string path = Path.Combine(OriginalDirectory, name.Name + ".dll");
            return File.Exists(path) ? owner.LoadFromAssemblyPath(path) : null;
        };
        string core = Path.Combine(OriginalDirectory, "Lokad.Onnx.dll");
        Assert.Equal("49901366484570493b7a42028e5c5458f30fe2c1703a01335b68d9a5f9a1fea9", Hash(core));
        return context.LoadFromAssemblyPath(core);
    }

    static string Hash(string path) => Convert.ToHexStringLower(SHA256.HashData(File.ReadAllBytes(path)));
    static float[] Values(int count, uint seed)
    {
        var a = new float[count];
        for (int i = 0; i < count; i++)
        {
            seed ^= seed << 13; seed ^= seed >> 17; seed ^= seed << 5;
            a[i] = ((seed & 0xffffff) / 8388608f - 1f) * .73f;
        }
        return a;
    }
    static DenseTensor<float> Data(int[] shape, uint seed = 8171) =>
        new DenseTensor<float>(Values(shape.Aggregate(1, (a,b) => a*b), seed), shape);

    static object OldTensor(Tensor<float> input)
    {
        object value = OriginalDense.GetMethod("OfShape")!.Invoke(null, new object[] { input.Dimensions.ToArray() })!;
        var buffer = (Memory<float>)OriginalDense.GetProperty("Buffer")!.GetValue(value)!;
        input.ToDenseTensor().Buffer.Span.CopyTo(buffer.Span);
        return value;
    }

    static void EqualBits(float[] expected, float[] actual)
    {
        Assert.Equal(expected.Length, actual.Length);
        for (int i = 0; i < expected.Length; i++)
        {
            if (float.IsNaN(expected[i])) { Assert.True(float.IsNaN(actual[i]), $"NaN at {i}"); continue; }
            Assert.True(BitConverter.SingleToInt32Bits(expected[i]) == BitConverter.SingleToInt32Bits(actual[i]),
                $"index {i}: original={expected[i]:R}, candidate={actual[i]:R}");
        }
    }

    static DenseTensor<float> Check(Tensor<float> x, Tensor<float> w, Tensor<float>? b,
        int group, int[] pads, int[] strides, int[]? dilation = null, TensorExecutionOptions? selected = null,
        bool expectDirect = true)
    {
        var options = selected ?? TensorExecutionOptions.Auto;
        var savedX = x.ToArray(); var savedW = w.ToArray(); var savedB = b?.ToArray();
        object oldOptions = Activator.CreateInstance(OriginalOptions,
            new object[] { options.UseSimd, options.UseIntrinsics, options.MaxDegreeOfParallelism })!;
        if (options.UseSegmentedConvolution)
            OriginalOptions.GetProperty("UseSegmentedConvolution", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(oldOptions, true);
        object expected = OriginalConv.Invoke(null, new object?[] { OldTensor(x), OldTensor(w), group, pads,
            b is null ? null : OldTensor(b), null, strides, dilation, oldOptions })!;
        var reference = ((Memory<float>)OriginalDense.GetProperty("Buffer")!.GetValue(expected)!).ToArray();
        var scratch = new ScratchAccountant();
        var result = (DenseTensor<float>)Tensor<float>.Conv2D(x, w, group, pads, b, null, strides, dilation,
            options with { ScratchReporter = scratch });
        EqualBits(reference, result.Buffer.ToArray());
        EqualBits(savedX, x.ToArray()); EqualBits(savedW, w.ToArray());
        if (b is not null) EqualBits(savedB!, b.ToArray());
        if (expectDirect && options.UseIntrinsics && Avx2.IsSupported) Assert.Equal(0, scratch.TotalScratchBytes);
        else Assert.True(scratch.TotalScratchBytes > 0);
        return result;
    }

    [Fact]
    public void ActualRuntimeAndProductsMatch()
    {
        Assert.True(OperatingSystem.IsLinux());
        Assert.Equal(".NET 10.0.8", RuntimeInformation.FrameworkDescription);
        Assert.Equal(1, Environment.ProcessorCount);
        Assert.Equal(4, Process.GetCurrentProcess().ProcessorAffinity.ToInt64());
        string? mode = Environment.GetEnvironmentVariable("DEPTHWISE_MODE");
        Assert.Contains(mode, new[] { "normal", "scalar" });
        Assert.Equal(mode == "normal", Fma.IsSupported && Avx2.IsSupported);
        Assert.Equal(Environment.GetEnvironmentVariable("DEPTHWISE_CORE_SHA"), Hash(typeof(ComputationalGraph).Assembly.Location));
        Assert.Equal(Environment.GetEnvironmentVariable("DEPTHWISE_DATA_SHA"), Hash(typeof(ParakeetTranscriber).Assembly.Location));
        Assert.Equal(Environment.GetEnvironmentVariable("DEPTHWISE_TEST_SHA"), Hash(typeof(DirectDepthwiseTests).Assembly.Location));
    }

    [Fact]
    public void EveryObservedGeometryMatchesOriginalBits()
    {
        using var census = JsonDocument.Parse(File.ReadAllText(Environment.GetEnvironmentVariable("DEPTHWISE_CENSUS")!));
        var records = new List<object>(); long values = 0;
        foreach (var row in census.RootElement.GetProperty("rows").EnumerateArray())
        {
            int[] g = row.GetProperty("geometry").EnumerateArray().Select(v => v.GetInt32()).ToArray();
            var x = Data(g[..4]); var w = Data(new[] { g[4], 1, g[5], g[6] }, 1729); var b = Data(new[] { g[4] }, 6131);
            var output = Check(x, w, b, g[17], g[11..15], g[9..11], g[7..9]);
            Assert.Equal(new[] { g[0], g[4], g[15], g[16] }, output.Dimensions.ToArray());
            values += output.Length;
            records.Add(new { geometry = g, checked_values = output.Length, bitwise_equal = true });
        }
        Assert.Equal(59, records.Count);
        string mode = Environment.GetEnvironmentVariable("DEPTHWISE_MODE")!;
        using var stream = new FileStream(Path.Combine(Environment.GetEnvironmentVariable("DEPTHWISE_EVIDENCE")!,
            "geometry-" + mode + ".json"), FileMode.CreateNew);
        JsonSerializer.Serialize(stream, new { passed = true, geometries = records.Count, checked_values = values,
            original_core_sha256 = Hash(Original.Location), records });
    }

    [Fact]
    public void SpatialBordersAndFlattenedVectorTailsMatch()
    {
        foreach (int h in new[] { 1, 2, 3, 7, 8, 9, 17, 18, 19 })
        for (int width = 1; width <= 35; width++)
            Check(Data(new[] { 1, 2, h, width }), Data(new[] { 2, 1, 3, 3 }, 13),
                width % 2 == 0 ? null : Data(new[] { 2 }, 29), 2, new[] { 1, 1, 1, 1 }, new[] { 2, 2 });
    }

    [Fact]
    public void LineTailsAndMemoryOffsetsMatch()
    {
        for (int outputs = 1; outputs <= 65; outputs++)
        {
            var storage = Values(3 * (outputs + 8) + 10, 17);
            var x = new DenseTensor<float>(storage.AsMemory(5, 3 * (outputs + 8)), new[] { 1, 3, 1, outputs + 8 });
            var weights = Values(33, 29);
            var w = new DenseTensor<float>(weights.AsMemory(3, 27), new[] { 3, 1, 1, 9 });
            Check(x, w, outputs % 2 == 0 ? null : Data(new[] { 3 }, 31), 3, new int[4], new[] { 1, 1 });
        }
    }

    [Fact]
    public void SpecialValuesIncludingPaddingMatch()
    {
        float[] special = [-0f, 0f, float.Epsilon, -float.Epsilon, float.MaxValue, -float.MaxValue,
            float.PositiveInfinity, float.NegativeInfinity, float.NaN];
        foreach (float value in special)
        {
            var x = Data(new[] { 1, 2, 11, 17 }); var w = Data(new[] { 2, 1, 3, 3 });
            x.Buffer.Span[0] = value; x.Buffer.Span[35] = value; w.Buffer.Span[0] = value;
            Check(x, w, null, 2, new[] { 1, 1, 1, 1 }, new[] { 2, 2 });
            x = Data(new[] { 1, 2, 1, 29 }); w = Data(new[] { 2, 1, 1, 9 });
            x.Buffer.Span[0] = value; x.Buffer.Span[28] = value; w.Buffer.Span[8] = value;
            Check(x, w, Data(new[] { 2 }), 2, new int[4], new[] { 1, 1 });
        }
    }

    [Fact]
    public void UnsupportedOptionsAndGeometriesKeepOriginalResults()
    {
        var x = Data(new[] { 1, 3, 1, 41 }); var w = Data(new[] { 3, 1, 1, 9 });
        foreach (var options in new[] { TensorExecutionOptions.Scalar, TensorExecutionOptions.Simd,
            TensorExecutionOptions.Auto with { MaxDegreeOfParallelism = 2 },
            TensorExecutionOptions.Auto with { UseSegmentedConvolution = true } })
            Check(x, w, null, 3, new int[4], new[] { 1, 1 }, selected: options, expectDirect: false);
        Check(x, w, null, 3, new int[4], new[] { 1, 2 }, expectDirect: false);
        Check(x, w, null, 3, new int[4], new[] { 1, 1 }, new[] { 1, 2 }, expectDirect: false);
        Check(x, w, null, 3, new[] { 0, 1, 0, 1 }, new[] { 1, 1 }, expectDirect: false);
        Check(Data(new[] { 2, 3, 1, 41 }), w, null, 3, new int[4], new[] { 1, 1 }, expectDirect: false);
        Check(x, Data(new[] { 3, 1, 1, 7 }), null, 3, new int[4], new[] { 1, 1 }, expectDirect: false);
        Check(x, Data(new[] { 3, 3, 1, 9 }), null, 1, new int[4], new[] { 1, 1 }, expectDirect: false);
    }

    [Fact]
    public void LogicalViewsAndPublicOutputsAreIndependent()
    {
        var source = Data(new[] { 1, 3, 1, 82 });
        var slice = new TensorSlice<float>(source, new[] { new SliceIndex(0, 1), new SliceIndex(0, 3),
            new SliceIndex(0, 1), new SliceIndex(0, 82, 2) });
        var reverse = new DenseTensor<float>(new[] { 1, 3, 1, 41 }, true);
        for (int c = 0; c < 3; c++) for (int j = 0; j < 41; j++) reverse[0,c,0,j] = source[0,c,0,j*2];
        var broadcast = Data(new[] { 1, 1, 1, 41 }).BroadcastDim(1, 3);
        var weights = Data(new[] { 3, 1, 1, 9 });
        foreach (var x in new Tensor<float>[] { slice, reverse, broadcast })
        {
            var first = Check(x, weights, null, 3, new int[4], new[] { 1, 1 });
            var saved = first.Buffer.ToArray();
            var second = Check(x, weights, null, 3, new int[4], new[] { 1, 1 });
            second.Buffer.Span.Fill(91f); EqualBits(saved, first.Buffer.ToArray());
        }
    }

    [Fact]
    public void ProviderOneDimensionalAndPooledSpatialOutputsMatch()
    {
        var x = Data(new[] { 1, 3, 1, 41 }); var w = Data(new[] { 3, 1, 1, 9 }); var b = Data(new[] { 3 });
        var expected = Check(x, w, b, 3, new int[4], new[] { 1, 1 });
        var result = CPUExecutionProvider.Conv(x.Reshape(1,3,41), w.Reshape(3,1,9), b, null,
            null, 3, null, new[] { 0, 0 }, new[] { 1 }, null);
        Assert.Equal(OpStatus.Success, result.Status);
        EqualBits(expected.Buffer.ToArray(), ((Tensor<float>)result.Outputs![0]).ToArray());
        x = Data(new[] { 1, 3, 17, 19 }); w = Data(new[] { 3, 1, 3, 3 });
        expected = Check(x, w, b, 3, new[] { 1, 1, 1, 1 }, new[] { 2, 2 });
        var pool = new TensorBufferPool(); int length = (int)expected.Length;
        var dirty = pool.Rent<float>(length); Array.Fill(dirty, float.NaN); pool.Return(dirty);
        var output = (DenseTensor<float>)Tensor<float>.Conv2D(x, w, 3, new[] { 1, 1, 1, 1 }, b,
            null, new[] { 2, 2 }, null, TensorExecutionOptions.Auto, pool);
        EqualBits(expected.Buffer.ToArray(), output.Buffer.ToArray());
        Assert.Equal(1, pool.Reused);
        Assert.True(MemoryMarshal.TryGetArray(output.Buffer, out var memory) && ReferenceEquals(dirty, memory.Array));
    }
}
